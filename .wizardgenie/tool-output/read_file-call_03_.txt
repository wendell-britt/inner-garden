31|     /**
32|      * Update player position and state
33|      * @param {number} deltaTime
34|      * @param {{x: number, y: number}} movement - Direction vector
35|      * @param {Array} obstacles - Collision rectangles
36|      */
37|     update(deltaTime, movement, obstacles = []) {
38|         this.isMoving = movement.x !== 0 || movement.y !== 0;
39|         
40|         if (this.currentAction) {
41|             this.actionTimer -= deltaTime;
42|             if (this.actionTimer <= 0) {
43|                 this.currentAction = null;
44|             }
45|             // Can't move while performing action
46|             return;
47|         }
48|         
49|         if (this.isMoving) {
50|             // Update direction based on movement
51|             if (Math.abs(movement.x) > Math.abs(movement.y)) {
52|                 this.direction = movement.x > 0 ? 'right' : 'left';
53|                 this.facing = movement.x > 0 ? 'east' : 'west';
54|             } else {
55|                 this.direction = movement.y > 0 ? 'down' : 'up';
56|                 this.facing = movement.y > 0 ? 'south' : 'north';
57|             }
58|             
59|             // Calculate new position
60|             let newX = this.x + movement.x * this.speed * (deltaTime / 1000);
61|             let newY = this.y + movement.y * this.speed * (deltaTime / 1000);
62|             
63|             // Collision check with obstacles
64|             let canMoveX = true;
65|             let canMoveY = true;
66|             
67|             for (const obs of obstacles) {
68|                 // Check X movement
69|                 if (this._checkCollision(newX, this.y, obs)) {
70|                     canMoveX = false;
71|                 }
72|                 // Check Y movement
73|                 if (this._checkCollision(this.x, newY, obs)) {
74|                     canMoveY = false;
75|                 }
76|             }
77|             
78|             if (canMoveX) this.x = newX;
79|             if (canMoveY) this.y = newY;
80|             
81|             // Animation
82|             this.animationTimer += deltaTime;
83|             if (this.animationTimer >= this.animationSpeed) {
84|                 this.animationTimer = 0;
85|                 this.animationFrame = (this.animationFrame + 1) % 4;
86|             }
87|             
88|             // Trail effect
89|             this.trailTimer += deltaTime;
90|             if (this.trailTimer >= 200) {
91|                 this.trailTimer = 0;
92|                 this.trailPositions.push({ x: this.x, y: this.y, life: 1 });
93|                 if (this.trailPositions.length > 10) {
94|                     this.trailPositions.shift();
95|                 }
96|             }
97|         } else {
98|             this.animationFrame = 0;
99|             this.animationTimer = 0;
100|         }
101|         
102|         // Update trail fade
103|         for (let i = this.trailPositions.length - 1; i >= 0; i--) {
104|             this.trailPositions[i].life -= deltaTime / 1000;
105|             if (this.trailPositions[i].life <= 0) {
106|                 this.trailPositions.splice(i, 1);
107|             }
108|         }
109|     }

115|         return px < obstacle.x + obstacle.w &&

117|                py < obstacle.y + obstacle.h &&

138|     /**
139|      * Get the player's interaction point (in front of them)
140|      */
141|     getInteractionPoint() {
142|         const range = this.interactionRange;
143|         switch (this.facing) {
144|             case 'north': return { x: this.x + this.width / 2, y: this.y - range };
145|             case 'south': return { x: this.x + this.width / 2, y: this.y + this.height + range };
146|             case 'east': return { x: this.x + this.width + range, y: this.y + this.height / 2 };
147|             case 'west': return { x: this.x - range, y: this.y + this.height / 2 };
148|             default: return { x: this.x + this.width / 2, y: this.y + this.height + range };
149|         }
150|     }

152|     /**
153|      * Check if a point is within interaction range
154|      */
155|     canInteractWith(targetX, targetY) {
156|         const dx = (this.x + this.width / 2) - targetX;
157|         const dy = (this.y + this.height / 2) - targetY;
158|         return Math.sqrt(dx * dx + dy * dy) <= this.interactionRange;
